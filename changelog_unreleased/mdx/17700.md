#### Fix multiline comments in MDX (#NNNN by @user)

<!-- Input -->
{/*

with blank lines

*/}

 * JSDoc-style comment
 */}

{/_

with blank lines

_/}

{/*

Multiline comment

with blank lines

*/}

{/**
 * JSDoc-style comment
 */}
